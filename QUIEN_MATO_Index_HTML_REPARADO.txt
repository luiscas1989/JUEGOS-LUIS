async function refresh(room,mode){
    if(busy)return;
    try{
      const auth=mode==='host'
        ? {hostKey:getHostKey(room),playerToken:getPlayerToken(room)}
        : (mode==='screen'?{hostKey:getHostKey(room)}:(mode==='player'?{playerToken:getPlayerToken(room)}:{}));
      STATE=await call('apiGetState',room,auth);
      await hydrateStateAssets(STATE);
      if(mode==='host')renderHost(STATE);else if(mode==='screen')renderScreen(STATE);else renderPlayer(STATE);

      if((mode==='host'||mode==='screen')&&['HOST_PROFILE','WAITING'].includes(STATE.status)&&STATE.prepStatus==='PENDING'){
        warmStory(STATE);
      }
      if((mode==='host'||mode==='screen')&&STATE.status==='WAITING'){
        warmCase(STATE);
      }
    }catch(e){toast(e.message,'error',7000)}
  }

function startRequirementText(s){
    if(s.players.length<s.minPlayers)return `Faltan ${s.minPlayers-s.players.length} jugador(es).`;
    if(!s.avatarsReady)return 'Espera a que terminen de convertirse todas las fotos en caricaturas.';
    if(!s.blueprintReady)return 'Gemini todavía está preparando el guion base.';
    if(!s.casePrepared)return 'La IA está adaptando el guion al reparto y eligiendo víctima/asesino.';
    if(s.casePrepError)return `El caso está listo, pero una imagen falló al precargarse. Puedes iniciar; el juego intentará continuar con las escenas disponibles.`;
    if((s.scenesPrepared||0)===0)return 'El caso ya está armado. Preparando la primera imagen de la historia…';
    if((s.scenesPrepared||0)<s.totalRounds)return `Caso listo. Se están precargando escenas (${s.scenesPrepared||0}/${s.totalRounds}); ya puedes iniciar.`;
    return 'Todo listo: guion, caricaturas y escenas preparadas.';
  }

function warmStory(s){
    if(!s||!['HOST_PROFILE','WAITING'].includes(s.status)||s.prepStatus!=='PENDING')return;
    const key=s.room+':story';if(STORY_WARMED.has(key))return;STORY_WARMED.add(key);
    google.script.run
      .withSuccessHandler(r=>{
        if(r&&r.warning)toast('El guion se preparó con respaldo local: '+r.warning,'warn',7000);
        setTimeout(()=>{if(STATE&&STATE.room===s.room)refresh(s.room,BOOT.view==='screen'?'screen':'host')},350)
      })
      .withFailureHandler(e=>{
        STORY_WARMED.delete(key);
        toast('No se pudo preparar el guion todavía: '+(e&&e.message?e.message:e),'warn',7000)
      })
      .apiPrepareStory(s.room,getHostKey(s.room));
  }

function warmCase(s){
    if(!s||s.status!=='WAITING'||!s.isHost)return;
    if(s.players.length<s.minPlayers||!s.avatarsReady||!s.blueprintReady)return;

    const rosterKey=s.players.map(p=>`${p.playerId}:${p.avatarFileId||''}`).sort().join('|');
    if(rosterKey!==LAST_ROSTER_KEY){
      LAST_ROSTER_KEY=rosterKey;
      LAST_ROSTER_AT=Date.now();
      return;
    }

    // Espera unos segundos sin cambios para evitar gastar imágenes si todavía están entrando jugadores.
    if(Date.now()-LAST_ROSTER_AT<6500||CASE_BUSY)return;

    CASE_BUSY=true;
    const action=!s.casePrepared?'apiPrepareCase':(((s.scenesPrepared||0)<s.totalRounds&&s.casePrepStatus!=='ERROR')?'apiPrepareCaseStep':'');
    if(!action){CASE_BUSY=false;return}

    google.script.run
      .withSuccessHandler(r=>{
        CASE_BUSY=false;
        if(r&&r.error)toast('Una escena no pudo prepararse todavía: '+r.error,'warn',7000);
        setTimeout(()=>{if(STATE&&STATE.room===s.room)refresh(s.room,BOOT.view==='screen'?'screen':'host')},250);
      })
      .withFailureHandler(e=>{
        CASE_BUSY=false;
        toast('La preparación continúa, pero hubo un fallo temporal: '+(e&&e.message?e.message:e),'warn',6000);
      })[action](s.room,getHostKey(s.room));
  }

  function screenControls(s){

    if(s.status==='ROUND'&&s.round<s.totalRounds)controls=`<button class="btn primary big" onclick="screenNextRound('${s.room}')">Siguiente ronda</button>`;
    if(s.status==='ROUND'&&s.round===s.totalRounds)controls=`<button class="btn gold big" onclick="screenOpenVoting('${s.room}')">Abrir votación final</button>`;
    if(s.status==='VOTING')controls=`<button class="btn primary big" onclick="screenReveal('${s.room}')">Revelar asesino</button>`;
    if(s.status==='ROUND')controls+=` <button class="btn ghost" onclick="narrate('${s.room}',${s.round||1})">🔊 Narrar</button>`;
    controls+=` <button class="btn danger" onclick="screenCloseRoom('${s.room}')">Cerrar sala</button>`;
    return `<div style="display:flex;gap:8px;flex-wrap:wrap;justify-content:center;margin-top:12px">${controls}</div>`;
  }
