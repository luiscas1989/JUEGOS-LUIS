function doGet(e) {
  setup_();
  const p = (e && e.parameter) || {};
  const t = HtmlService.createTemplateFromFile('Index');
  t.bootJson = JSON.stringify({
    view: String(p.view || '').toLowerCase(),
    room: normalizeRoom_(p.room || ''),
    hostKey: String(p.hostKey || ''),
    version: QM.VERSION,
    appUrl: getAppUrl_(),
  });
  return t.evaluate()
    .setTitle('¿Quién mató a...?')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, viewport-fit=cover');
}

function include(filename) {
  return HtmlService.createHtmlOutputFromFile(filename).getContent();
}

function diagnosticoApis() {
  setup_();
  const cfg = getConfig_();
  const out = {
    ok: true,
    version: QM.VERSION,
    gemini: { configured: !!cfg.geminiKey, ok: false, structuredOk: false, ttsOk: false, detail: '' },
    cloudflare: {
      configured: !!(cfg.cfToken && cfg.cfAccountId),
      ok: false,
      detail: '',
      missingAccountId: !cfg.cfAccountId,
    },
  };

  if (!cfg.geminiKey) {
    out.ok = false;
    out.gemini.detail = 'Falta GEMINI_API_KEY en Script Properties.';
  } else {
    try {
      const txt = geminiText_('Responde únicamente con la palabra OK.', { maxOutputTokens: 16, temperature: 0 });
      out.gemini.ok = /^OK\b/i.test(String(txt).trim());
      const sj = geminiJson_('Devuelve un objeto JSON con ok=true y mensaje="OK".', {
        type: 'object',
        properties: { ok: { type: 'boolean' }, mensaje: { type: 'string' } },
        required: ['ok','mensaje'],
      });
      out.gemini.structuredOk = !!(sj && sj.ok === true);
      const ttsBlob = geminiTts_('Prueba de narración.', 0);
      out.gemini.ttsOk = !!(ttsBlob && ttsBlob.getBytes().length > 100);
      out.gemini.ok = !!(out.gemini.ok && out.gemini.structuredOk && out.gemini.ttsOk);
      out.gemini.detail = out.gemini.ok ? 'Texto, JSON estructurado y TTS de Gemini respondieron correctamente.' : 'Gemini respondió, pero alguna capacidad no pasó la prueba.';
    } catch (err) {
      out.ok = false;
      out.gemini.detail = safeError_(err);
    }
  }

  if (!cfg.cfToken || !cfg.cfAccountId) {
    out.ok = false;
    out.cloudflare.detail = !cfg.cfAccountId
      ? 'Falta CLOUDFLARE_ACCOUNT_ID. El token por sí solo no basta.'
      : 'Falta CLOUDFLARE_API_TOKEN.';
  } else {
    try {
      const blob = cloudflareImage_({
        prompt: 'simple noir detective evidence card, one brass key on a dark table, comic 3D illustration, no text',
        width: 256,
        height: 256,
        inputBlobs: [],
      });
      out.cloudflare.ok = blob && blob.getBytes().length > 1000;
      out.cloudflare.detail = out.cloudflare.ok ? 'Cloudflare Workers AI generó una imagen de prueba.' : 'Cloudflare respondió sin una imagen válida.';
    } catch (err) {
      out.ok = false;
      out.cloudflare.detail = safeError_(err);
    }
  }

  out.ok = !!(out.gemini.ok && out.cloudflare.ok);
  return out;
}

  try {

      room,
      hostKey,
      hostName,
      status: 'SETUP',
      round: 0,
      createdAt: new Date().toISOString(),

      victimId: '',
      killerId: '',
      storyJson: '',
      sceneAssetIdsJson: '{}',
      audioAssetIdsJson: '{}',
      resultJson: '',
      configJson: '{}',
      blueprintJson: '',

/**
 * Paso 1: finaliza la configuración y abre la sala a jugadores INMEDIATAMENTE.
 * La historia se prepara en una segunda llamada (apiPrepareStory) para que el
 * anfitrión no tenga que esperar a Gemini antes de mostrar el QR.
 */

function apiFinalizeSetup(room, hostKey, config) {
  setup_();
  room = normalizeRoom_(room);
  const clean = normalizeGameConfig_(config || {});
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const r = requireHost_(room, hostKey);
    if (r.status !== 'SETUP') throw new Error('La configuración de esta sala ya fue finalizada.');
    updateRoom_(room, {
      hostName: '',
      status: 'HOST_PROFILE',
      configJson: JSON.stringify(clean),
      blueprintJson: '',
      prepStatus: 'PENDING',
      prepError: '',
      caseRosterSignature: '',
      casePrepStatus: 'IDLE',
      casePrepError: '',
      updatedAt: new Date().toISOString(),
    });
    logEvent_(room, 'SETUP_FINALIZED', { config: clean, next: 'HOST_PROFILE' });
    return { ok: true, room, status: 'HOST_PROFILE', prepStatus: 'PENDING' };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Paso 2: prepara la historia base en una ejecución independiente.
 * Puede llamarse desde el móvil del anfitrión o desde Ver partida.
 * Un lock evita generar el caso dos veces si ambas pantallas lo piden a la vez.
 */
function apiPrepareStory(room, hostKey) {
  setup_();
  room = normalizeRoom_(room);
  let clean;
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const r = requireHost_(room, hostKey);
    if (['HOST_PROFILE','WAITING'].indexOf(String(r.status)) < 0) return { ok: false, skipped: true, reason: 'STATUS' };
    if (String(r.prepStatus || '') === 'READY' && String(r.blueprintJson || '').trim()) {
      return { ok: true, skipped: true, prepStatus: 'READY' };
    }
    if (String(r.prepStatus || '') === 'GENERATING') {
      return { ok: true, skipped: true, prepStatus: 'GENERATING' };
    }
    clean = normalizeGameConfig_(parseJson_(r.configJson, {}));
    updateRoom_(room, { prepStatus: 'GENERATING', prepError: '', updatedAt: new Date().toISOString() });
    logEvent_(room, 'BLUEPRINT_GENERATING', {});
  } finally {
    lock.releaseLock();
  }

  let blueprint;
  let warning = '';
  let aiMode = 'GEMINI';
  try {
    blueprint = generateBlueprint_(room, clean);
  } catch (err) {
    aiMode = 'FALLBACK';
    warning = safeError_(err);
    blueprint = fallbackBlueprint_(clean);
    logEvent_(room, 'BLUEPRINT_FALLBACK', { error: warning });
  }

  updateRoom_(room, {
    blueprintJson: JSON.stringify(normalizeBlueprint_(blueprint, clean)),
    prepStatus: 'READY',
    prepError: warning,
    updatedAt: new Date().toISOString(),
  });
  logEvent_(room, 'BLUEPRINT_READY', { aiMode });
  return { ok: true, room, warning, prepStatus: 'READY' };
}


/**
 * Registra al anfitrión como primer jugador DESPUÉS de guardar la configuración.
 * El QR general permanece oculto hasta que su avatar esté preparado.
 */
function apiJoinHost(room, hostKey, name, photoDataUrl) {
  setup_();
  room = normalizeRoom_(room);
  name = cleanName_(name);
  if (!room || !name) throw new Error('El nombre del anfitrión es obligatorio.');
  if (!photoDataUrl || !/^data:image\//i.test(photoDataUrl)) throw new Error('Debes tomar o seleccionar una foto.');

  const photoBlob = dataUrlToBlob_(photoDataUrl, 'foto-anfitrion.jpg');
  if (photoBlob.getBytes().length > 1800000) throw new Error('La foto es demasiado grande. Inténtalo de nuevo.');

  let playerId = '';
  let token = '';
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const r = requireHost_(room, hostKey);

    // Si ya se completó el alta del anfitrión, devuelve su sesión en vez de duplicarlo.
    const existing = getPlayers_(room, true).find(p => String(p.role || '') === 'HOST_PENDING');
    if (r.status === 'WAITING' && existing && existing.avatarFileId) {
      return { ok: true, room, playerId: existing.playerId, token: existing.token, avatarMode: String(existing.avatarStatus || 'READY'), reused: true };
    }
    if (r.status !== 'HOST_PROFILE') throw new Error('La sala no está esperando el perfil del anfitrión.');

    if (existing) {
      playerId = existing.playerId;
      token = existing.token;
      updatePlayer_(room, playerId, {
        name,
        avatarStatus: 'PROCESSING',
        avatarError: '',
        active: true,
      });
    } else {
      playerId = 'P_' + randomToken_(10);
      token = randomToken_(28);
      appendObject_(getSheet_(QM.SHEET_PLAYERS), PLAYER_HEADERS, {
        room,
        playerId,
        token,
        name,
        avatarFileId: '',
        joinedAt: new Date().toISOString(),
        role: 'HOST_PENDING',
        secret: '',
        alibi: '',
        vote: '',
        score: 0,
        active: true,
        avatarStatus: 'PROCESSING',
        avatarError: '',
      });
    }
    updateRoom_(room, { hostName: name, updatedAt: new Date().toISOString() });
    logEvent_(room, 'HOST_PROFILE_STARTED', { playerId, name });
    SpreadsheetApp.flush();
  } finally {
    lock.releaseLock();
  }

  let avatarMode = 'AI';
  let warning = '';
  let file;
  try {
    const avatarBlob = makeAvatar_(photoBlob, name);
    file = saveAsset_(avatarBlob, room + '-avatar-' + playerId);
    updatePlayer_(room, playerId, {
      avatarFileId: file.getId(),
      avatarStatus: 'READY',
      avatarError: '',
    });
  } catch (err) {
    avatarMode = 'PHOTO_FALLBACK';
    warning = 'No se pudo crear la caricatura ahora. Se usará la foto temporalmente: ' + safeError_(err);
    const fallback = photoBlob.copyBlob().setName('avatar-' + playerId + '.jpg');
    file = saveAsset_(fallback, room + '-avatar-' + playerId);
    updatePlayer_(room, playerId, {
      avatarFileId: file.getId(),
      avatarStatus: 'FALLBACK',
      avatarError: warning,
    });
  }

  updateRoom_(room, {
    hostName: name,
    status: 'WAITING',
    storyJson: '',
    sceneAssetIdsJson: '{}',
    audioAssetIdsJson: '{}',
    resultJson: '',
    victimId: '',
    killerId: '',
    caseRosterSignature: '',
    casePrepStatus: 'PENDING',
    casePrepError: '',
    updatedAt: new Date().toISOString(),
  });
  logEvent_(room, 'HOST_READY', { playerId, name, avatarMode });
  SpreadsheetApp.flush();
  return { ok: true, room, playerId, token, avatarMode, warning };
}

function apiJoin(room, name, photoDataUrl) {
  setup_();
  room = normalizeRoom_(room);
  name = cleanName_(name);
  if (!room || !name) throw new Error('Código de sala y nombre son obligatorios.');
  if (!photoDataUrl || !/^data:image\//i.test(photoDataUrl)) throw new Error('Debes tomar o seleccionar una foto.');

  const photoBlob = dataUrlToBlob_(photoDataUrl, 'foto.jpg');
  if (photoBlob.getBytes().length > 1800000) throw new Error('La foto es demasiado grande. Inténtalo de nuevo.');

  const playerId = 'P_' + randomToken_(10);
  const token = randomToken_(28);

  // Alta rápida: el jugador aparece en Ver partida mientras la IA crea el avatar.
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const r = requireRoom_(room);
    if (r.status === 'SETUP') throw new Error('El anfitrión todavía está configurando la partida.');
    if (r.status === 'HOST_PROFILE') throw new Error('El anfitrión está creando su personaje. Espera a que aparezca el QR general.');
    if (r.status !== 'WAITING') throw new Error('La partida ya comenzó.');
    const players = getPlayers_(room, true);
    if (players.length >= QM.MAX_PLAYERS) throw new Error('La sala ya tiene el máximo de 12 jugadores.');
    if (players.some(p => normalizeText_(p.name) === normalizeText_(name))) {
      throw new Error('Ya existe un jugador con ese nombre.');
    }

    appendObject_(getSheet_(QM.SHEET_PLAYERS), PLAYER_HEADERS, {
      room,
      playerId,
      token,
      name,
      avatarFileId: '',
      joinedAt: new Date().toISOString(),
      role: '',
      secret: '',
      alibi: '',
      vote: '',
      score: 0,
      active: true,
      avatarStatus: 'PROCESSING',
      avatarError: '',
    });
    invalidatePreparedCase_(room, 'PLAYER_JOINED');
    logEvent_(room, 'PLAYER_JOINED', { playerId, name, avatarStatus: 'PROCESSING' });
  } finally {
    lock.releaseLock();
  }

  let avatarMode = 'AI';
  let warning = '';
  let file;
  try {
    const avatarBlob = makeAvatar_(photoBlob, name);
    file = saveAsset_(avatarBlob, room + '-avatar-' + playerId);
    updatePlayer_(room, playerId, {
      avatarFileId: file.getId(),
      avatarStatus: 'READY',
      avatarError: '',
    });
  } catch (err) {
    // Respaldo: la partida continúa aunque Cloudflare falle temporalmente.
    avatarMode = 'PHOTO_FALLBACK';
    warning = 'No se pudo crear la caricatura ahora. Se usará tu foto temporalmente: ' + safeError_(err);
    const fallback = photoBlob.copyBlob().setName('avatar-' + playerId + '.jpg');
    file = saveAsset_(fallback, room + '-avatar-' + playerId);
    updatePlayer_(room, playerId, {
      avatarFileId: file.getId(),
      avatarStatus: 'FALLBACK',
      avatarError: warning,
    });
  }

  touchRoom_(room);
  logEvent_(room, 'AVATAR_READY', { playerId, name, avatarMode });
  return { ok: true, room, playerId, token, avatarMode, warning };
}

function apiGetState(room, auth) {
  setup_();
  room = normalizeRoom_(room);
  const r = requireRoom_(room);
  const players = getPlayers_(room, true);
  const story = parseJson_(r.storyJson, null);
  const result = parseJson_(r.resultJson, null);
  const scenes = parseJson_(r.sceneAssetIdsJson, {});
  const audios = parseJson_(r.audioAssetIdsJson, {});
  const config = parseJson_(r.configJson, {});
  const blueprintReady = !!String(r.blueprintJson || '').trim() && String(r.prepStatus || '') === 'READY';
  const avatarsReady = players.length > 0 && players.every(p => !!p.avatarFileId && String(p.avatarStatus || 'READY') !== 'PROCESSING');
  const rosterSignature = rosterSignature_(players);
  const casePrepared = !!String(r.storyJson || '').trim() &&
    !!String(r.victimId || '').trim() &&
    !!String(r.killerId || '').trim() &&
    String(r.caseRosterSignature || '') === rosterSignature;
  const scenesPrepared = Array.from({ length: QM.TOTAL_ROUNDS }, (_, i) => String(scenes[String(i + 1)] || ''))
    .filter(id => id && id.indexOf('__GEN__') !== 0).length;

  const victimPublic = ['ROUND','VOTING','REVEAL'].indexOf(String(r.status)) >= 0;
  const publicPlayers = players.map(p => ({
    playerId: p.playerId,
    name: p.name,
    avatarFileId: p.avatarFileId,
    avatarStatus: String(p.avatarStatus || (p.avatarFileId ? 'READY' : 'PROCESSING')),
    score: Number(p.score || 0),
    active: truthy_(p.active),
    isVictim: victimPublic && r.victimId === p.playerId,
    voteSubmitted: !!p.vote,
  }));

  let privatePlayer = null;
  let isHost = false;
  auth = auth || {};
  if (auth.playerToken) {
    const me = players.find(p => p.token === String(auth.playerToken));
    if (me) {
      privatePlayer = {
        playerId: me.playerId,
        name: me.name,
        role: me.role || '',
        secret: me.secret || '',
        alibi: me.alibi || '',
        vote: me.vote || '',
        score: Number(me.score || 0),
        avatarStatus: String(me.avatarStatus || ''),
      };
    }
  }
  if (auth.hostKey && String(auth.hostKey) === String(r.hostKey)) isHost = true;

  const currentRound = story && Number(r.round) > 0 ? story.rounds[Number(r.round) - 1] : null;
  const voteCounts = (r.status === 'REVEAL' || isHost) ? countVotes_(players) : null;

  return {
    ok: true,
    room,
    hostName: r.hostName,
    status: r.status,
    round: Number(r.round || 0),
    totalRounds: QM.TOTAL_ROUNDS,
    players: publicPlayers,
    currentRound: sanitizeRound_(currentRound),
    caseTitle: story ? story.caseTitle : '',
    location: story ? story.location : '',
    victimId: victimPublic ? (r.victimId || '') : '',
    sceneFileId: (String(scenes[String(r.round)] || '').indexOf('__GEN__') === 0 ? '' : (scenes[String(r.round)] || '')),
    audioFileId: audios[String(r.round)] || '',
    result: r.status === 'REVEAL' ? result : null,
    voteCounts,
    privatePlayer,
    isHost,
    links: isHost ? roomLinks_(room, r.hostKey) : null,
    canStart: isHost && r.status === 'WAITING' && players.length >= QM.MIN_PLAYERS && avatarsReady && blueprintReady && casePrepared && (scenesPrepared >= 1 || !!String(r.casePrepError || '')),
    minPlayers: QM.MIN_PLAYERS,
    maxPlayers: QM.MAX_PLAYERS,
    avatarsReady,
    blueprintReady,
    casePrepared,
    scenesPrepared,
    casePrepStatus: String(r.casePrepStatus || 'IDLE'),
    casePrepError: isHost ? String(r.casePrepError || '') : '',
    prepStatus: String(r.prepStatus || 'IDLE'),
    prepError: isHost ? String(r.prepError || '') : '',
    config: isHost ? config : null,
    updatedAt: r.updatedAt,
  };
}

function apiGetAsset(fileId) {
  setup_();
  fileId = String(fileId || '');
  if (!fileId) return null;
  const file = DriveApp.getFileById(fileId);
  if (!isGameAsset_(file)) throw new Error('Recurso no autorizado.');
  const blob = file.getBlob();
  return {
    fileId,
    mimeType: blob.getContentType() || 'application/octet-stream',
    base64: Utilities.base64Encode(blob.getBytes()),
  };
}


/**
 * Prepara el reparto definitivo del caso durante la sala de espera.
 * Se ejecuta cuando hay al menos 4 jugadores, todas las caricaturas están listas
 * y Gemini ya generó el guion base. Si entra alguien nuevo, se invalida y se rehace.
 */
function apiPrepareCase(room, hostKey) {
  setup_();
  room = normalizeRoom_(room);
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const r = requireHost_(room, hostKey);
    if (r.status !== 'WAITING') return { ok: false, skipped: true, reason: 'STATUS' };

    const players = getPlayers_(room, true);
    if (players.length < QM.MIN_PLAYERS) return { ok: false, skipped: true, reason: 'MIN_PLAYERS' };
    if (players.some(p => !p.avatarFileId || String(p.avatarStatus || '') === 'PROCESSING')) {
      return { ok: false, skipped: true, reason: 'AVATARS' };
    }

    const blueprint = parseJson_(r.blueprintJson, null);
    if (!blueprint || String(r.prepStatus || '') !== 'READY') {
      return { ok: false, skipped: true, reason: 'BLUEPRINT' };
    }

    const signature = rosterSignature_(players);
    const scenes = parseJson_(r.sceneAssetIdsJson, {});
    const sameCast = String(r.caseRosterSignature || '') === signature;
    const storyReady = sameCast && !!String(r.storyJson || '').trim() && !!String(r.victimId || '') && !!String(r.killerId || '');

    if (storyReady) {
      const count = countPreparedScenes_(scenes);
      const nextStatus = count >= QM.TOTAL_ROUNDS ? 'READY' : 'SCENES';
      if (String(r.casePrepStatus || '') !== nextStatus) {
        updateRoom_(room, { casePrepStatus: nextStatus, updatedAt: new Date().toISOString() });
      }
      return { ok: true, prepared: true, scenesPrepared: count, casePrepStatus: nextStatus };
    }

    const shuffled = shuffle_(players.slice());
    const victim = shuffled[0];
    const killer = shuffled[1];
    const story = materializeStory_(blueprint, players, victim, killer);

    updateRoom_(room, {
      victimId: victim.playerId,
      killerId: killer.playerId,
      storyJson: JSON.stringify(story),
      sceneAssetIdsJson: '{}',
      audioAssetIdsJson: '{}',
      resultJson: '',
      caseRosterSignature: signature,
      casePrepStatus: 'SCENES',
      casePrepError: '',
      updatedAt: new Date().toISOString(),
    });
    logEvent_(room, 'CASE_PREPARED', {
      rosterSize: players.length,
      victimId: victim.playerId,
      killerId: killer.playerId,
    });

    return { ok: true, prepared: true, scenesPrepared: 0, casePrepStatus: 'SCENES' };
  } finally {
    lock.releaseLock();
  }
}

/**
 * Genera UNA escena por llamada para repartir la carga mientras los jugadores esperan.
 * Host y Ver pantalla pueden invocarla; un marcador temporal evita duplicados.
 */
function apiPrepareCaseStep(room, hostKey) {
  setup_();
  room = normalizeRoom_(room);

  let targetRound = 0;
  let expectedSignature = '';

  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const r = requireHost_(room, hostKey);
    if (r.status !== 'WAITING') return { ok: false, skipped: true, reason: 'STATUS' };

    const players = getPlayers_(room, true);
    const signature = rosterSignature_(players);
    expectedSignature = signature;

    if (!String(r.storyJson || '').trim() || String(r.caseRosterSignature || '') !== signature) {
      return { ok: false, skipped: true, reason: 'CASE_NOT_READY' };
    }

    const scenes = parseJson_(r.sceneAssetIdsJson, {});
    for (let round = 1; round <= QM.TOTAL_ROUNDS; round++) {
      const value = String(scenes[String(round)] || '');
      if (isPreparedAssetId_(value)) continue;

      if (value.indexOf('__GEN__:') === 0) {
        const ts = Number(value.split(':')[1] || 0);
        if (ts && (Date.now() - ts) < 120000) {
          return { ok: true, skipped: true, busy: true, scenesPrepared: countPreparedScenes_(scenes) };
        }
      }

      targetRound = round;
      scenes[String(round)] = '__GEN__:' + Date.now();
      updateRoom_(room, {
        sceneAssetIdsJson: JSON.stringify(scenes),
        casePrepStatus: 'SCENES',
        casePrepError: '',
        updatedAt: new Date().toISOString(),
      });
      break;
    }

    if (!targetRound) {
      updateRoom_(room, {
        casePrepStatus: 'READY',
        casePrepError: '',
        updatedAt: new Date().toISOString(),
      });
      return { ok: true, ready: true, scenesPrepared: QM.TOTAL_ROUNDS, casePrepStatus: 'READY' };
    }
  } finally {
    lock.releaseLock();
  }

  let fileId = '';
  let generationError = '';
  try {
    fileId = generateRoundSceneFile_(room, targetRound);
  } catch (err) {
    generationError = safeError_(err);
  }

  const lock2 = LockService.getScriptLock();
  lock2.waitLock(20000);
  try {
    const r2 = requireHost_(room, hostKey);
    const players2 = getPlayers_(room, true);
    const currentSignature = rosterSignature_(players2);

    // Si entró alguien durante la generación, no mezclar una escena del reparto antiguo.
    if (r2.status !== 'WAITING' ||
        String(r2.caseRosterSignature || '') !== expectedSignature ||
        currentSignature !== expectedSignature) {
      if (fileId) {
        try { DriveApp.getFileById(fileId).setTrashed(true); } catch (e) {}
      }
      return { ok: true, stale: true };
    }

    const scenes2 = parseJson_(r2.sceneAssetIdsJson, {});
    if (generationError) {
      delete scenes2[String(targetRound)];
      updateRoom_(room, {
        sceneAssetIdsJson: JSON.stringify(scenes2),
        casePrepStatus: 'ERROR',
        casePrepError: generationError,
        updatedAt: new Date().toISOString(),
      });
      logEvent_(room, 'CASE_SCENE_ERROR', { round: targetRound, error: generationError });
      return { ok: false, round: targetRound, error: generationError, scenesPrepared: countPreparedScenes_(scenes2) };
    }

    scenes2[String(targetRound)] = fileId;
    const count = countPreparedScenes_(scenes2);
    const done = count >= QM.TOTAL_ROUNDS;
    updateRoom_(room, {
      sceneAssetIdsJson: JSON.stringify(scenes2),
      casePrepStatus: done ? 'READY' : 'SCENES',
      casePrepError: '',
      updatedAt: new Date().toISOString(),
    });
    logEvent_(room, 'CASE_SCENE_READY', { round: targetRound, scenesPrepared: count });
    return { ok: true, round: targetRound, fileId, scenesPrepared: count, casePrepStatus: done ? 'READY' : 'SCENES' };
  } finally {
    lock2.releaseLock();
  }
}


function apiStart(room, hostKey) {
  setup_();
  room = normalizeRoom_(room);

  let needFirstScene = false;

  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const r = requireHost_(room, hostKey);
    if (r.status !== 'WAITING') throw new Error('La sala todavía no está lista para iniciar.');

    const players = getPlayers_(room, true);
    if (players.length < QM.MIN_PLAYERS) throw new Error('Se necesitan al menos ' + QM.MIN_PLAYERS + ' jugadores.');
    if (players.some(p => !p.avatarFileId || String(p.avatarStatus || '') === 'PROCESSING')) {
      throw new Error('Todavía hay caricaturas procesándose. Espera unos segundos.');
    }

    const signature = rosterSignature_(players);
    let story = parseJson_(r.storyJson, null);
    let victim = players.find(p => p.playerId === String(r.victimId || ''));
    let killer = players.find(p => p.playerId === String(r.killerId || ''));
    const preparedForThisCast = story && victim && killer && String(r.caseRosterSignature || '') === signature;

    if (!preparedForThisCast) {
      let blueprint = parseJson_(r.blueprintJson, null);
      const config = parseJson_(r.configJson, {});
      if (!blueprint) {
        blueprint = fallbackBlueprint_(config);
        logEvent_(room, 'BLUEPRINT_MISSING_FALLBACK', {});
      }
      const shuffled = shuffle_(players.slice());
      victim = shuffled[0];
      killer = shuffled[1];
      story = materializeStory_(blueprint, players, victim, killer);
      updateRoom_(room, {
        victimId: victim.playerId,
        killerId: killer.playerId,
        storyJson: JSON.stringify(story),
        sceneAssetIdsJson: '{}',
        caseRosterSignature: signature,
        casePrepStatus: 'SCENES',
        casePrepError: '',
      });
    }

    const latest = requireRoom_(room);
    const scenes = parseJson_(latest.sceneAssetIdsJson, {});
    needFirstScene = !isPreparedAssetId_(String(scenes['1'] || ''));

    updateRoom_(room, {
      status: 'ROUND',
      round: 1,
      audioAssetIdsJson: '{}',
      resultJson: '',
      updatedAt: new Date().toISOString(),
    });

    applySecrets_(room, story.secrets);
    clearVotes_(room);
    logEvent_(room, 'GAME_STARTED', {
      victimId: victim.playerId,
      killerId: killer.playerId,
      storyMode: preparedForThisCast ? 'PRELOADED' : 'ON_START',
    });
  } finally {
    lock.releaseLock();
  }

  let assetWarning = '';
  if (needFirstScene) {
    try { ensureRoundScene_(room, 1); } catch (err) {
      assetWarning = safeError_(err);
      logEvent_(room, 'SCENE_ERROR', { round: 1, error: assetWarning });
    }
  }
  return { ok: true, room, round: 1, warning: assetWarning };
}

function apiWarmScene(room, hostKey, round) {
  setup_();
  room = normalizeRoom_(room);
  round = Number(round || 0);
  const r = requireHost_(room, hostKey);
  if (r.status !== 'ROUND') return { ok: false, skipped: true };
  if (round < 1 || round > QM.TOTAL_ROUNDS) return { ok: false, skipped: true };
  if (round <= Number(r.round || 0)) return { ok: true, skipped: true };
  const fileId = ensureRoundScene_(room, round);
  return { ok: true, round, fileId };
}

function apiNextRound(room, hostKey) {
  setup_();
  room = normalizeRoom_(room);
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  let next;
  try {
    const r = requireHost_(room, hostKey);
    if (r.status !== 'ROUND') throw new Error('No se puede avanzar desde el estado actual.');
    const current = Number(r.round || 0);
    if (current >= QM.TOTAL_ROUNDS) throw new Error('Ya estás en la última ronda. Abre la votación.');
    next = current + 1;
    updateRoom_(room, { round: next, updatedAt: new Date().toISOString() });
    logEvent_(room, 'ROUND_CHANGED', { round: next });
  } finally {
    lock.releaseLock();
  }

  let warning = '';
  try { ensureRoundScene_(room, next); } catch (err) { warning = safeError_(err); logEvent_(room, 'SCENE_ERROR', { round: next, error: warning }); }
  return { ok: true, room, round: next, warning };
}

function apiVote(room, playerToken, suspectId) {
  setup_();
  room = normalizeRoom_(room);
  suspectId = String(suspectId || '');
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const r = requireRoom_(room);
    if (r.status !== 'VOTING') throw new Error('La votación no está abierta.');
    const players = getPlayers_(room, true);
    const me = players.find(p => p.token === String(playerToken || ''));
    if (!me) throw new Error('Sesión de jugador no válida.');
    if (!players.some(p => p.playerId === suspectId)) throw new Error('Sospechoso no válido.');
    if (me.playerId === suspectId) throw new Error('No puedes votarte a ti mismo.');
    updatePlayer_(room, me.playerId, { vote: suspectId });
    touchRoom_(room);
    logEvent_(room, 'VOTE', { voter: me.playerId });
    return { ok: true };
  } finally {
    lock.releaseLock();
  }
}

function apiReveal(room, hostKey) {
  setup_();
  room = normalizeRoom_(room);
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const r = requireHost_(room, hostKey);
    if (r.status !== 'VOTING') throw new Error('Primero abre la votación.');
    const players = getPlayers_(room, true);
    const story = parseJson_(r.storyJson, {});
    const killer = players.find(p => p.playerId === r.killerId);
    const victim = players.find(p => p.playerId === r.victimId);
    const counts = countVotes_(players);
    const correct = players.filter(p => p.vote === r.killerId);
    const votedCount = players.filter(p => !!p.vote).length;
    const killerEscaped = correct.length < Math.ceil(Math.max(votedCount, 1) / 2);

    players.forEach(p => {
      let delta = 0;
      if (p.vote === r.killerId) delta += 100;
      if (p.playerId === r.killerId && killerEscaped) delta += 150;
      if (p.playerId === r.victimId && p.vote === r.killerId) delta += 50;
      if (delta) updatePlayer_(room, p.playerId, { score: Number(p.score || 0) + delta });
    });

    const result = {
      killerId: r.killerId,
      killerName: killer ? killer.name : 'Desconocido',
      victimId: r.victimId,
      victimName: victim ? victim.name : 'Desconocido',
      motive: story.finalReveal && story.finalReveal.motive || 'Un secreto que llevaba demasiado tiempo enterrado.',
      method: story.finalReveal && story.finalReveal.method || 'El crimen fue preparado para desviar todas las sospechas.',
      explanation: story.finalReveal && story.finalReveal.explanation || 'Las pistas encajaban cuando se revisaron las contradicciones.',
      finalLine: story.finalReveal && story.finalReveal.finalLine || 'La verdad siempre deja una huella.',
      voteCounts: counts,
      correctPlayerIds: correct.map(p => p.playerId),
      killerEscaped,
    };
    updateRoom_(room, { status: 'REVEAL', resultJson: JSON.stringify(result), updatedAt: new Date().toISOString() });
    logEvent_(room, 'REVEAL', { killerId: r.killerId, correct: result.correctPlayerIds });
    return { ok: true, result };
  } finally {
    lock.releaseLock();
  }
}

function apiNarrate(room, round) {
  setup_();
  room = normalizeRoom_(room);
  round = Number(round || 0);
  const r = requireRoom_(room);
  const story = parseJson_(r.storyJson, null);
  if (!story || !story.rounds || !story.rounds[round - 1]) throw new Error('Ronda no disponible.');
  const audios = parseJson_(r.audioAssetIdsJson, {});
  if (audios[String(round)]) return { ok: true, fileId: audios[String(round)], cached: true };

  const narration = String(story.rounds[round - 1].narration || '').slice(0, 1600);
  const blob = geminiTts_(narration, round);
  const file = saveAsset_(blob, room + '-audio-r' + round);
  audios[String(round)] = file.getId();
  updateRoom_(room, { audioAssetIdsJson: JSON.stringify(audios), updatedAt: new Date().toISOString() });
  logEvent_(room, 'AUDIO_CREATED', { round });
  return { ok: true, fileId: file.getId(), cached: false };
}

  const storyStyles = ['La Mujer de Judas','Caso original','Novia enmascarada','Secretos de familia'];
  let location = String(config.location || 'Aleatoria');
  let mood = String(config.mood || 'Telenovela de misterio');
  let storyStyle = String(config.storyStyle || 'La Mujer de Judas');
  if (locations.indexOf(location) < 0) location = 'Aleatoria';
  if (moods.indexOf(mood) < 0) mood = 'Telenovela de misterio';

  return {
    location,
    mood,
    storyStyle,
    rounds: QM.TOTAL_ROUNDS,
    artStyle: 'Cómic 3D cinematográfico',
  };
}

function materializeStory_(blueprint, players, victim, killer) {
  const bp = normalizeBlueprint_(blueprint, {});
  const pub = s => replaceCaseTokens_(s, victim.name, '');
  const fin = s => replaceCaseTokens_(s, victim.name, killer.name);
  const rounds = bp.rounds.map((r, i) => ({
    n: i + 1,
    title: pub(r.title),
    narration: pub(r.narration),
    publicClue: pub(r.publicClue),
    evidenceTitle: pub(r.evidenceTitle),
    evidenceText: pub(r.evidenceText),
    imagePrompt: pub(r.imagePrompt),
    featuredPlayerIds: featuredPlayersForRound_(players, victim, i + 1),
  }));
  const secrets = buildPrivateSecrets_(players, victim, killer, {
    motive: fin(bp.finalReveal.motive),
    location: pub(bp.location),
  });
  return {
    caseTitle: pub(bp.caseTitle),
    location: pub(bp.location),
    synopsis: pub(bp.synopsis),
    rounds,
    secrets,
    finalReveal: {
      motive: fin(bp.finalReveal.motive),
      method: fin(bp.finalReveal.method),
      explanation: fin(bp.finalReveal.explanation),
      finalLine: fin(bp.finalReveal.finalLine),
    },
  };
}

function replaceCaseTokens_(value, victimName, killerName) {
  let s = String(value || '');
  s = s.replace(/\[VICTIM\]/gi, victimName || 'la víctima');
  s = s.replace(/\[KILLER\]/gi, killerName || 'una persona del grupo');
  return s;
}

function featuredPlayersForRound_(players, victim, round) {
  const others = players.filter(p => p.playerId !== victim.playerId);
  if (!others.length) return [victim.playerId];
  const picked = [];
  const start = ((round - 1) * 3) % others.length;
  for (let i = 0; i < Math.min(3, others.length); i++) picked.push(others[(start + i) % others.length].playerId);
  if (round <= 2) picked.unshift(victim.playerId);
  return unique_(picked).slice(0, 4);
}

function buildPrivateSecrets_(players, victim, killer, ctx) {
  const motives = [
    'ocultaste una deuda que tenías con [VICTIM]',
    'borraste un mensaje que [VICTIM] te envió esa tarde',
    'discutiste con [VICTIM] por un secreto familiar',
    'entraste en una zona donde dijiste no haber estado',
    'conocías una información que [VICTIM] pensaba revelar',
    'mentiste sobre la hora en la que viste por última vez a [VICTIM]',
    'guardabas una fotografía que podía cambiar la investigación',
    'tenías una relación secreta con alguien del grupo',
    'encontraste una de las pruebas antes que los demás y no lo dijiste',
    'habías firmado un acuerdo privado con [VICTIM]',
    'escuchaste una conversación comprometida y decidiste callarla',
    'llevabas un objeto parecido a una de las evidencias',
  ];
  const shuffled = shuffle_(motives.slice());
  let j = 0;
  return players.map((p, idx) => {
    if (p.playerId === victim.playerId) {
      return {
        playerId: p.playerId, role: 'VICTIM',
        secret: 'Eres la víctima de este caso. Sigues participando como investigador fantasma: analiza las pistas y vota al final.',
        alibi: 'Tu objetivo es reconstruir qué ocurrió y descubrir quién te mató.',
      };
    }
    if (p.playerId === killer.playerId) {
      return {
        playerId: p.playerId, role: 'KILLER',
        secret: 'Tú eres quien mató a ' + victim.name + '. Tu motivo está relacionado con esto: ' + String(ctx.motive || 'un secreto que no podía salir a la luz') + '. Defiéndete sin confesar y aprovecha las dudas razonables.',
        alibi: 'Afirmas que durante el momento clave estabas lejos de la escena. Mantén tu versión mientras las pistas no la contradigan.',
      };
    }
    const seed = shuffled[j++ % shuffled.length].replace(/\[VICTIM\]/g, victim.name);
    return {
      playerId: p.playerId, role: 'SUSPECT',
      secret: 'Tu secreto: ' + seed + '. No eres el asesino, pero esta información puede hacerte parecer culpable.',
      alibi: idx % 2 ? 'Tu coartada tiene un intervalo de varios minutos que nadie puede confirmar.' : 'Dices haber estado con el grupo, aunque una parte de tu recorrido no tiene testigos.',
    };
  });
}

function geminiText_(prompt, opts) {
  const cfg = getConfig_();
  if (!cfg.geminiKey) throw new Error('GEMINI_API_KEY no configurada.');
  opts = opts || {};
  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + encodeURIComponent(cfg.geminiTextModel) + ':generateContent';
  const body = {
    contents: [{ role: 'user', parts: [{ text: String(prompt) }] }],
    generationConfig: {
      temperature: typeof opts.temperature === 'number' ? opts.temperature : 0.9,
      maxOutputTokens: opts.maxOutputTokens || 4096,
    },
  };
  const res = fetchWithRetry_(url, {
    method: 'post',
    contentType: 'application/json',
    headers: { 'x-goog-api-key': cfg.geminiKey },
    payload: JSON.stringify(body),
    muteHttpExceptions: true,
  }, 2);
  assertHttp_(res, 'Gemini');
  const json = JSON.parse(res.getContentText());
  const text = (((json.candidates || [])[0] || {}).content || {}).parts || [];
  return text.map(p => p.text || '').join('').trim();
}

function geminiJson_(prompt, schema) {
  const cfg = getConfig_();
  if (!cfg.geminiKey) throw new Error('GEMINI_API_KEY no configurada.');
  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + encodeURIComponent(cfg.geminiTextModel) + ':generateContent';
  const body = {
    contents: [{ role: 'user', parts: [{ text: String(prompt) }] }],
    generationConfig: {
      temperature: 1.05,
      maxOutputTokens: 10000,
      responseMimeType: 'application/json',
      responseSchema: normalizeGeminiSchema_(schema),
    },
  };
  const res = fetchWithRetry_(url, {
    method: 'post',
    contentType: 'application/json',
    headers: { 'x-goog-api-key': cfg.geminiKey },
    payload: JSON.stringify(body),
    muteHttpExceptions: true,
  }, 2);
  assertHttp_(res, 'Gemini');
  const json = JSON.parse(res.getContentText());
  const parts = ((((json.candidates || [])[0] || {}).content || {}).parts || []);
  const text = parts.map(p => p.text || '').join('').trim();
  if (!text) throw new Error('Gemini no devolvió contenido JSON.');
  try { return JSON.parse(text); } catch (err) { throw new Error('Gemini devolvió JSON no válido: ' + text.slice(0, 250)); }
}

  const r = requireRoom_(room);

function generateRoundSceneFile_(room, round) {
  const r = requireRoom_(room);
  const story = parseJson_(r.storyJson, null);
  if (!story || !story.rounds || !story.rounds[round - 1]) throw new Error('No existe la ronda ' + round + '.');

  const players = getPlayers_(room, true);
  const rr = story.rounds[round - 1];
  let ids = Array.isArray(rr.featuredPlayerIds) ? rr.featuredPlayerIds.slice(0, 4) : [];
  if (round <= 2 && r.victimId && ids.indexOf(r.victimId) < 0) ids.unshift(r.victimId);
  ids = unique_(ids).filter(id => players.some(p => p.playerId === id)).slice(0, 4);

  const refs = ids.map(id => players.find(p => p.playerId === id)).filter(Boolean);
  const blobs = refs.map(p => DriveApp.getFileById(p.avatarFileId).getBlob());
  const refLegend = refs.map((p, i) => 'input image ' + i + ' is ' + p.name).join('; ');

  const prompt = [
    'Create a cinematic 16:9 still frame for a multiplayer murder mystery game, premium 3D comic / graphic-novel style.',
    'Latin dramatic thriller atmosphere, expressive lighting, visual clues, elegant composition suitable for a TV.',
    refLegend ? ('Character references: ' + refLegend + '. Preserve their recognizable avatar faces and design.') : '',
    String(rr.imagePrompt || ''),
    'No written words, no captions, no logos. No explicit gore. If a victim is shown, keep the scene tasteful and non-graphic.',
  ].filter(Boolean).join(' ');

  const blob = cloudflareImage_({
    prompt,
    width: QM.SCENE_WIDTH,
    height: QM.SCENE_HEIGHT,
    inputBlobs: blobs,
  }).setName(room + '-scene-r' + round + '.png');

  const file = saveAsset_(blob, room + '-scene-r' + round);
  return file.getId();
}

function cloudflareImage_(opts) {
  const cfg = getConfig_();
  if (!cfg.cfToken) throw new Error('CLOUDFLARE_API_TOKEN no configurado.');
  if (!cfg.cfAccountId) throw new Error('CLOUDFLARE_ACCOUNT_ID no configurado.');
  opts = opts || {};
  const model = cfg.cfImageModel;
  const url = 'https://api.cloudflare.com/client/v4/accounts/' + encodeURIComponent(cfg.cfAccountId) + '/ai/run/' + model;
  const payload = {
    prompt: String(opts.prompt || '3D comic mystery illustration'),
    width: String(opts.width || 512),
    height: String(opts.height || 512),
  };
  (opts.inputBlobs || []).slice(0, 4).forEach((blob, i) => {
    payload['input_image_' + i] = blob;
  });

  const res = fetchWithRetry_(url, {
    method: 'post',
    headers: { Authorization: 'Bearer ' + cfg.cfToken },
    payload,
    muteHttpExceptions: true,
  }, 2);
  assertHttp_(res, 'Cloudflare Workers AI');

  const contentType = String((res.getHeaders()['Content-Type'] || res.getHeaders()['content-type'] || '')).toLowerCase();
  if (contentType.indexOf('image/') === 0) return res.getBlob().setName('generated.png');

  const text = res.getContentText();
  let json;
  try { json = JSON.parse(text); } catch (err) { throw new Error('Cloudflare devolvió una respuesta no JSON y no imagen.'); }
  if (json.success === false) throw new Error('Cloudflare: ' + JSON.stringify(json.errors || json.messages || []).slice(0, 500));
  let b64 = null;
  if (json.result && typeof json.result.image === 'string') b64 = json.result.image;
  else if (json.result && typeof json.result === 'string') b64 = json.result;
  else if (json.image && typeof json.image === 'string') b64 = json.image;
  if (!b64) throw new Error('Cloudflare no devolvió el campo de imagen esperado.');
  b64 = b64.replace(/^data:image\/[^;]+;base64,/, '');
  return Utilities.newBlob(Utilities.base64Decode(b64), 'image/png', 'generated.png');
}

function normalizeRounds_(rounds, players, victim, killer) {
  const ids = players.map(p => p.playerId);
  const fallback = fallbackStory_(players, victim, killer).rounds;
  if (!Array.isArray(rounds)) rounds = [];
  const out = [];
  for (let i = 0; i < QM.TOTAL_ROUNDS; i++) {
    const src = rounds[i] || fallback[i];
    const featured = unique_((src.featuredPlayerIds || []).map(String).filter(id => ids.indexOf(id) >= 0)).slice(0, 4);
    if (!featured.length) featured.push(victim.playerId);
    out.push({
      n: i + 1,
      title: String(src.title || fallback[i].title).slice(0, 100),
      narration: String(src.narration || fallback[i].narration).slice(0, 1800),
      publicClue: String(src.publicClue || fallback[i].publicClue).slice(0, 500),
      evidenceTitle: String(src.evidenceTitle || fallback[i].evidenceTitle).slice(0, 120),
      evidenceText: String(src.evidenceText || fallback[i].evidenceText).slice(0, 600),
      imagePrompt: String(src.imagePrompt || fallback[i].imagePrompt).slice(0, 1300),
      featuredPlayerIds: featured,
    });
  }
  return out;
}

function normalizeSecrets_(secrets, players, victim, killer) {
  secrets = Array.isArray(secrets) ? secrets : [];
  return players.map(p => {
    const s = secrets.find(x => String(x.playerId) === p.playerId) || {};
    let role = p.playerId === killer.playerId ? 'KILLER' : (p.playerId === victim.playerId ? 'VICTIM' : 'SUSPECT');
    let secret = String(s.secret || 'Ocultas un detalle de aquella noche que podría hacerte parecer culpable.').slice(0, 700);
    let alibi = String(s.alibi || 'Tu coartada contiene un pequeño hueco de tiempo.').slice(0, 500);
    if (role === 'VICTIM') {
      secret = 'Eres la víctima de este caso, pero sigues jugando como investigador fantasma. Observa las pistas y vota al final.';
      alibi = 'No necesitas defender una coartada: reconstruye lo que ocurrió.';
    }
    return { playerId: p.playerId, role, secret, alibi };
  });
}

function setup_() {
  const db = getDb_();
  ensureSheet_(db, QM.SHEET_ROOMS, ROOM_HEADERS);
  ensureSheet_(db, QM.SHEET_PLAYERS, PLAYER_HEADERS);

function getDb_() {
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty(QM.PROP_DB_ID);
  if (id) {
    try { return SpreadsheetApp.openById(id); } catch (e) { props.deleteProperty(QM.PROP_DB_ID); }
  }
  let ss = null;
  try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch (e) {}
  if (!ss) ss = SpreadsheetApp.create('QUIEN MATO - BD');
  props.setProperty(QM.PROP_DB_ID, ss.getId());
  return ss;
}

function getAssetFolder_() {
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty(QM.PROP_ASSET_FOLDER);
  if (id) {
    try { return DriveApp.getFolderById(id); } catch (e) { props.deleteProperty(QM.PROP_ASSET_FOLDER); }
  }
  const folder = DriveApp.createFolder('QUIEN_MATO_ASSETS');
  props.setProperty(QM.PROP_ASSET_FOLDER, folder.getId());
  return folder;
}

function ensureSheet_(ss, name, headers) {
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
  } else {
    const existingWidth = Math.max(1, sh.getLastColumn());
    const existing = sh.getRange(1, 1, 1, existingWidth).getValues()[0].map(String);
    const expectedPrefix = headers.slice(0, Math.min(existing.length, headers.length));
    const prefixOk = expectedPrefix.every((h, i) => existing[i] === h || existing[i] === '');
    if (!prefixOk) throw new Error('La hoja ' + name + ' tiene una estructura incompatible. Haz una copia antes de migrarla.');
    if (existing.length < headers.length) {
      sh.getRange(1, existing.length + 1, 1, headers.length - existing.length).setValues([headers.slice(existing.length)]);
    }
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
  }
  sh.setFrozenRows(1);
  return sh;
}

function getSheet_(name) { return getDb_().getSheetByName(name); }

function getRowsAsObjects_(sheet, headers) {
  const last = sheet.getLastRow();
  if (last < 2) return [];
  const values = sheet.getRange(2, 1, last - 1, headers.length).getValues();
  return values.map((row, idx) => {
    const o = { _row: idx + 2 };
    headers.forEach((h, i) => o[h] = row[i]);
    return o;
  });
}

function appendObject_(sheet, headers, obj) {
  sheet.appendRow(headers.map(h => obj[h] !== undefined ? obj[h] : ''));
}

function requireRoom_(room) {
  room = normalizeRoom_(room);
  const r = getRowsAsObjects_(getSheet_(QM.SHEET_ROOMS), ROOM_HEADERS)
    .find(x => sameRoom_(x.room, room));
  if (!r) throw new Error('Sala no encontrada.');
  return r;
}


function requireHost_(room, hostKey) {
  const r = requireRoom_(room);
  if (!hostKey || String(r.hostKey) !== String(hostKey)) throw new Error('Acceso de anfitrión no válido.');
  return r;
}

function updateRoom_(room, patch) {
  const sh = getSheet_(QM.SHEET_ROOMS);
  const r = requireRoom_(room);
  const values = ROOM_HEADERS.map(h => patch[h] !== undefined ? patch[h] : r[h]);
  sh.getRange(r._row, 1, 1, ROOM_HEADERS.length).setValues([values]);
}

function touchRoom_(room) { updateRoom_(room, { updatedAt: new Date().toISOString() }); }

function getPlayers_(room, activeOnly) {
  room = normalizeRoom_(room);
  let list = getRowsAsObjects_(getSheet_(QM.SHEET_PLAYERS), PLAYER_HEADERS)
    .filter(p => sameRoom_(p.room, room));
  if (activeOnly) list = list.filter(p => truthy_(p.active));
  return list;
}

function updatePlayer_(room, playerId, patch) {
  room = normalizeRoom_(room);
  const sh = getSheet_(QM.SHEET_PLAYERS);
  const p = getRowsAsObjects_(sh, PLAYER_HEADERS).find(x => sameRoom_(x.room, room) && x.playerId === playerId);
  if (!p) throw new Error('Jugador no encontrado.');
  const values = PLAYER_HEADERS.map(h => patch[h] !== undefined ? patch[h] : p[h]);
  sh.getRange(p._row, 1, 1, PLAYER_HEADERS.length).setValues([values]);
}

function saveAsset_(blob, baseName) {
  const folder = getAssetFolder_();

}

function isGameAsset_(file) {
  const folderId = getAssetFolder_().getId();
  const parents = file.getParents();
  while (parents.hasNext()) if (parents.next().getId() === folderId) return true;
  return false;
}

function normalizeGeminiSchema_(value) {
  if (Array.isArray(value)) return value.map(normalizeGeminiSchema_);
  if (!value || typeof value !== 'object') return value;
  const out = {};
  Object.keys(value).forEach(k => {
    let v = value[k];
    if (k === 'type' && typeof v === 'string') v = v.toUpperCase();
    out[k] = normalizeGeminiSchema_(v);
  });
  return out;
}

function dataUrlToBlob_(dataUrl, filename) {
  const m = String(dataUrl).match(/^data:([^;]+);base64,(.+)$/);
  if (!m) throw new Error('Formato de imagen no válido.');
  const mime = m[1];
  if (!/^image\/(jpeg|jpg|png|webp)$/i.test(mime)) throw new Error('Tipo de imagen no admitido.');
  return Utilities.newBlob(Utilities.base64Decode(m[2]), mime, filename || 'photo.jpg');
}

function extensionForMime_(mime) {
  mime = String(mime || '').toLowerCase();
  if (mime.indexOf('png') >= 0) return '.png';
  if (mime.indexOf('wav') >= 0) return '.wav';
  if (mime.indexOf('mpeg') >= 0 || mime.indexOf('mp3') >= 0) return '.mp3';
  if (mime.indexOf('webp') >= 0) return '.webp';
  return '.jpg';
}

function fetchWithRetry_(url, options, retries) {
  let last;
  for (let i = 0; i <= retries; i++) {
    try {
      const res = UrlFetchApp.fetch(url, options);
      const code = res.getResponseCode();
      if (code !== 429 && code < 500) return res;
      last = new Error('HTTP ' + code + ': ' + res.getContentText().slice(0, 500));
    } catch (err) { last = err; }
    if (i < retries) Utilities.sleep(1200 * (i + 1));
  }
  throw last || new Error('Error de red.');
}

function assertHttp_(res, label) {
  const code = res.getResponseCode();
  if (code >= 200 && code < 300) return;
  let body = '';
  try { body = res.getContentText().slice(0, 1000); } catch (e) {}
  throw new Error(label + ' HTTP ' + code + ': ' + body);
}

function pcmToWavBlob_(pcmBytes, sampleRate, channels, bitsPerSample, filename) {
  const dataLen = pcmBytes.length;
  const byteRate = sampleRate * channels * bitsPerSample / 8;
  const blockAlign = channels * bitsPerSample / 8;
  let h = [];
  h = h.concat(asciiBytes_('RIFF'));
  h = h.concat(le32_(36 + dataLen));
  h = h.concat(asciiBytes_('WAVE'));
  h = h.concat(asciiBytes_('fmt '));
  h = h.concat(le32_(16));
  h = h.concat(le16_(1));
  h = h.concat(le16_(channels));
  h = h.concat(le32_(sampleRate));
  h = h.concat(le32_(byteRate));
  h = h.concat(le16_(blockAlign));
  h = h.concat(le16_(bitsPerSample));
  h = h.concat(asciiBytes_('data'));
  h = h.concat(le32_(dataLen));
  const all = h.map(toSignedByte_).concat(pcmBytes);
  return Utilities.newBlob(all, 'audio/wav', filename || 'audio.wav');
}

function asciiBytes_(s) { return String(s).split('').map(c => c.charCodeAt(0)); }
function le16_(n) { return [n & 255, (n >>> 8) & 255]; }
function le32_(n) { return [n & 255, (n >>> 8) & 255, (n >>> 16) & 255, (n >>> 24) & 255]; }
function toSignedByte_(n) { n = n & 255; return n > 127 ? n - 256 : n; }
